# RV32I base integer instruction encodings.
# See ../encoding-format.md and ../validation.md.

instruction lui {
    encoding {
        bits[6:0] = 0b0110111
    }
    operands {
        rd: reg = bits[11:7]
        imm: bits<32> = concat(bits[31:12], 0b000000000000)
    }
}
